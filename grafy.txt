from typing import Any


class Vertex:
    data: Any

    def __init__(self, data=None):
        self.data = data

    def __str__(self):
        return str(self.data)


class Edge:
    vertices: set[Vertex]
    weight: float

    def __init__(self, set, weight=1.0):
        self.set = set
        self.weight = weight

    def __str__(self):
        for i in self.set:
            return str(i) + " - "


class Graph:
    vertices: list[Vertex]
    edges: set[Edge]

    def __init__(self, vertices=None, edges=None):
        self.vertices = vertices
        self.edges = edges

    def __str__(self):
        lista1 = []
        for i in self.vertices:
            lista1.append(i)
        return lista1

    def add_vertex(self, value):
        if self.vertices is not None:
            self.vertices.append(Vertex(value))
        else:
            self.vertices = []
            self.vertices.append(Vertex(value))


    def size(self):
        liczba = 0
        if self.vertices is not None:
            for i in self.vertices:
                liczba += 1
        return liczba

    def edge_size(self):
        liczba = 0
        if self.edges is not None:
            for i in self.edges:
                liczba += 1
        return liczba

    def add_edge(self, vertex1, vertex2, weight=1.0):
        if self.edges is not None:
            self.edges.add(Edge({vertex1, vertex2}))
        else:
            self.edges = {Edge({vertex1,vertex2})}

    def is_edge(self, vertex1, vertex2):
        if self.edges is not None:
            for i in self.edges:
                if i.set is not None:
                    for j in i.set:
                        if j == vertex1:
                            for u in i.set:
                                if u == vertex2:
                                    return True
        return False

    def cone(self, value):
        a = Vertex(value)
        if self.vertices is not None:
            for i in self.vertices:
                self.add_edge(a,i)
            self.vertices.append(a)
        else:
            self.vertices = []
            self.vertices.append(a)
    def complete_graph(self,*values):
        for i in values:
            self.add_vertex(i)
        for i in self.vertices:
            for j in self.vertices:
                if self.is_edge(i,j) == False and i != j:
                    self.add_edge(i,j)



G = Graph()
